Setup & Browser
Import & AUT constant
Setup
import { browser as vibium } from 'vibium'
import assert from 'node:assert/strict'

const AUT = process.env.AUT_BASE_URL
  || 'https://automation-exercise.daisyladybug.com'
Browser lifecycle
start / record / stop
// Local dev (visible browser)
const browser = await vibium.start({ headless: false })

// CI-aware
const browser = await vibium.start({
  headless: process.env.CI === 'true'
})

// Record (captures replay URL)
const { browser, stop } = await vibium.record({ headless: true })
// ... after browser.stop():
const url = await stop()
console.log('Recording:', url)

// Always clean up
try { /* test */ } finally { await browser.stop() }
Context & Navigation
Context and page
Hierarchy
const context = await browser.newContext()
const page    = await context.newPage()

// Two isolated sessions
const [ctx1, ctx2] = await Promise.all([
  browser.newContext(), browser.newContext()
])
const [page1, page2] = await Promise.all([
  ctx1.newPage(), ctx2.newPage()
])
Navigation
go / waitForURL / url / title
await page.go(`${AUT}/products`)
await page.go(`${AUT}/products/prod_001`)

// Wait for URL pattern after navigation
await page._waitForURL('**/confirmation')

const url   = await page.url()
const title = await page.title()
Finding Elements
page.find() and page.findAll()
Locators
// Role + text — preferred for interactive elements
const btn     = await page.find({ role: 'button', text: 'Add to Cart' })
const link    = await page.find({ role: 'link',   text: 'Products' })
const heading = await page.find({ role: 'heading', text: 'Products' })

// CSS selector — for inputs and structural targeting
const search   = await page.find("input[placeholder='Search products...']")
const category = await page.find('select:first-of-type')
const field    = await page.find('input[name=email]')

// Text only — returns outermost matching element; use role for interactive elements
const label = await page.find({ text: '1 item · 1 product' })

// All matching elements
const cards = await page.findAll('a[href*="/products/prod_"]')
const count = cards.length
State, Actions, Assertions
Element state methods
Read-only queries
await element.isVisible()  // rendered and on screen
await element.isEnabled()  // not disabled
await element.isChecked()  // checkbox/radio state

await element.text()       // visible text content
await element.value()      // current input value
await element.attr('href') // attribute value
Actions & assertions
Mutate + verify
await element.click()
await element.fill('text')            // clear then type
await element.type('text')            // append without clearing
await element.selectOption('Label')   // <select> dropdowns

assert.equal(actual, expected, 'label')
assert.ok(truthy, 'label')
assert.ok(await element.isVisible(), 'visible')
assert.equal(await page.title(), 'expected', 'title')
Waiting Patterns
Always wait before asserting on dynamic state
Action → wait → assert
// find() polls until element exists — use it as a wait
await page.find({ text: '2 items · 1 product' })

// Wait for URL after navigation
await page._waitForURL('**/confirmation')

// The pattern: trigger action → wait for observable outcome → assert
await button.click()
await page.find({ text: 'Success' })          // <— wait first
assert.ok(await banner.isVisible(), 'banner') // <— then assert
Captures
Promise.all + fire-and-forget — all three capture types
dialog / console / download
// Dialog — intercept and handle
const [result] = await Promise.all([
  page.capture.dialog('accept'),  // or 'dismiss'
  (async () => { trigger.click() })(),
])
// result.type ('alert' | 'confirm' | 'prompt')  result.message

// Console — collect messages
const [messages] = await Promise.all([
  page.capture.console(),
  (async () => { trigger.click() })(),
])
// messages[].type ('log' | 'warning' | 'error' | 'info')  messages[].text

// Download — save file
const [download] = await Promise.all([
  page.capture.download('/tmp/downloads'),
  (async () => { trigger.click() })(),
])
// download.suggestedFilename  download.path
Network Interception
Always register before page.go()
fulfill / abort / continue
// Stub a successful response
await page.route('**/api/products*', async route => {
  await route.fulfill({
    status: 200, contentType: 'application/json',
    body: JSON.stringify([{ id: 'prod_stub', name: 'Test', price: 1 }]),
  })
})

// Simulate HTTP error
await page.route('**/api/products*', async route => {
  await route.fulfill({ status: 500, body: '{"error":"Server Error"}' })
})

// Connection failure
await page.route('**/api/products*', async route => { await route.abort() })

// Partial intercept — stub one, pass everything else
await page.route('**/*', async route => {
  if (route.request().url().includes('/api/products')) {
    await route.fulfill({ status: 200, contentType: 'application/json', body: '[]' })
  } else {
    await route.continue()
  }
})
Test Structure
Vitest suite with lifecycle hooks
beforeEach / afterEach
import { browser as vibium } from 'vibium'
import { describe, it, expect,
         beforeEach, afterEach } from 'vitest'

let browser: any, page: any

beforeEach(async () => {
  browser = await vibium.start({
    headless: process.env.CI === 'true'
  })
  const context = await browser.newContext()
  page = await context.newPage()
})

afterEach(async () => { await browser.stop() })

describe('Products', () => {
  it('shows search input', async () => {
    await page.go(`${AUT}/products`)
    const input = await page.find(
      "input[placeholder='Search products...']"
    )
    expect(await input.isVisible()).toBe(true)
  })
})
vibium.record() wired into Vitest
stopRecording pattern
let stopRecording: (() => Promise<string>) | null = null

beforeEach(async () => {
  const rec = await vibium.record({
    headless: process.env.CI === 'true'
  })
  browser = rec.browser
  stopRecording = rec.stop
  const ctx = await browser.newContext()
  page = await ctx.newPage()
})

afterEach(async () => {
  await browser.stop()        // browser first
  if (stopRecording) {
    const url = await stopRecording()
    console.log('Recording:', url)
    stopRecording = null
  }
})
Page Object Pattern
Centralise selectors — keep tests selector-free
ProductsPage example
class ProductsPage {
  constructor(private page: Page) {}

  async goto() {
    await this.page.go(`${AUT}/products`)
  }
  async search(query: string) {
    const input = await this.page.find("input[placeholder='Search products...']")
    await input.fill(query)
  }
  async filterByCategory(category: string) {
    const select = await this.page.find('select:first-of-type')
    await select.selectOption(category)
  }
  async getProductCount() {
    return (await this.page.findAll('a[href*="/products/prod_"]')).length
  }
}

// In a test:
const products = new ProductsPage(page)
await products.goto()
await products.filterByCategory('Electronics')
const count = await products.getProductCount()
assert.equal(count, 3, '3 Electronics products')
CI Daemon (GitHub Actions)
start before tests · stop with if: always()
Daemon lifecycle in YAML
- name: Start Vibium daemon
  run: vibium daemon start
  env:
    HOME: /root

- name: Run tests
  run: npm run test:ci
  env:
    CI: true
    HOME: /root
    AUT_BASE_URL: ${{ vars.AUT_BASE_URL }}

- name: Stop Vibium daemon
  if: always()            # runs even if tests failed
  run: vibium daemon stop
  env:
    HOME: /root
